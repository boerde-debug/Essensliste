  }


  function renderSeason(season, listId) {

    var list = sortedDates().filter(function (x) { return x.data.season === season; });

    var listEl = document.getElementById(listId);

    var me = norm(getMyIdentity().name);

    if (!list.length) {

      listEl.innerHTML = '<p class="empty-state">Noch keine Termine eingetragen.</p>';

    } else {

      listEl.innerHTML = list.map(function (x) { return renderDateRow(x, me); }).join("");

    }

    bindDateRowHandlers(listEl);

  }


  function renderDateRow(x, myNorm) {

    var d = x.data;

    var signups = d.signups || [];

    var slotsHtml = [signups[0], signups[1]].map(function (s, i) {

      if (s) {

        var mine = norm(s.name) === myNorm && myNorm;

        return '<span class="slot filled' + (mine ? " mine" : "") + '" data-slot-click="' + x.id + '|' + i + '">' +

          esc(s.name) + (s.phone ? ' <span class="phone">· ' + esc(s.phone) + "</span>" : "") + "</span>";

      }

      return '<span class="slot empty" data-slot-click="' + x.id + '|' + i + '">+ Eintragen</span>';

    }).join("");

    var delBtn = isOwner ? '<button class="admin-del" data-del-date="' + x.id + '">✕</button>' : "";

    return '<div class="date-row' + (d.needsReplacement ? " flagged" : "") + '">' +

      '<div class="date-block"><div class="day">' + dayNum(x.id) + '</div><div class="mon">' + esc(monShort(x.id)) + "</div></div>" +

      '<div class="date-meta"><div class="title-line">' + esc(d.label || "Training") + '</div>' +

      '<div class="sub-line">' + esc(longDate(x.id)) + (d.time ? " · " + esc(d.time) : "") + '</div>' +

      '<div class="slots">' + slotsHtml + "</div></div>" + delBtn + "</div>";

  }


  function bindDateRowHandlers(container) {

    container.querySelectorAll("[data-slot-click]").forEach(el => {

      el.onclick = function () {

        var parts = el.getAttribute("data-slot-click").split("|");

        var dateId = parts[0], idx = parseInt(parts[1], 10);

        var entry = (datesById[dateId].data.signups || [])[idx];

        var me = norm(getMyIdentity().name);

        if (entry && me && norm(entry.name) === me) {

          openWithdrawModal(dateId, entry.name);

        } else if (!entry) {

          openSignupModal(dateId);

        } else {

          toast("Belegt von " + entry.name);

        }

      };

    });

    container.querySelectorAll("[data-del-date]").forEach(el => {

      el.onclick = async function () {

        var id = el.getAttribute("data-del-date");

        showConfirm("Termin wirklich löschen?", async () => {

          await supabaseClient.from('dates').delete().eq('id', id);

          toast("Termin gelöscht.");

        }, "Löschen");

      };

    });

  }


  function renderMine() {

    var ident = getMyIdentity();

    var me = norm(ident.name);

    var el = document.getElementById("mineContent");

    if (!me) { el.innerHTML = '<p class="empty-state">Gib oben deinen Namen ein.</p>'; return; }

    var mine = [];

    sortedDates().forEach(x => {

      (x.data.signups || []).forEach(s => { if (norm(s.name) === me) mine.push(x); });

    });

    var winter = mine.filter(x => x.data.season === "winter").length;

    var sommer = mine.filter(x => x.data.season === "sommer").length;

    el.innerHTML = mine.length ? mine.map(m => '<div class="mine-item"><span>' + esc(longDate(m.id)) + '</span></div>').join("") : '<p class="empty-state">Keine Termine.</p>';

    el.innerHTML += '<div class="quota-row"><span class="quota-pill ' + (winter >= 1 ? "ok" : "open") + '">Winter ' + winter + '/1</span><span class="quota-pill ' + (sommer >= 2 ? "ok" : "open") + '">Sommer ' + sommer + '/2</span></div>';

  }


  function renderOverview() {

    var byName = {};

    sortedDates().forEach(x => {

      (x.data.signups || []).forEach(s => {

        var k = norm(s.name);

        if (!k) return;

        if (!byName[k]) byName[k] = { name: s.name, winter: 0, sommer: 0 };

        byName[k][x.data.season === "winter" ? "winter" : "sommer"]++;

      });

    });

    var rows = Object.values(byName).sort((a,b) => a.name.localeCompare(b.name));

    var html = "<thead><tr><th>Name</th><th>Winter</th><th>Sommer</th></tr></thead><tbody>";

    rows.forEach(r => { html += "<tr><td>" + esc(r.name) + "</td><td>" + r.winter + "</td><td>" + r.sommer + "</td></tr>"; });

    document.getElementById("overviewCard").querySelector(".wrap-scroll").innerHTML = '<table class="overview">' + html + "</tbody></table>";

  }


  function renderHistory() {

    var entries = [];

    sortedDates().forEach(x => { (x.data.history || []).forEach(h => entries.push(h)); });

    entries.sort((a,b) => b.ts.localeCompare(a.ts));

    document.getElementById("historyList").innerHTML = entries.length ? entries.map(e => '<div class="history-entry"><span class="ts">' + fmtTs(e.ts) + '</span><span>' + esc(e.text) + '</span></div>').join("") : '<p class="empty-state">Keine Einträge</p>';

  }


  async function addDate(season) {

    var dateVal = document.getElementById("addDate_" + season).value;

    var label = document.getElementById("addLabel_" + season).value;

    var from = document.getElementById("addFrom_" + season).value;

    var to = document.getElementById("addTo_" + season).value;

    if (!dateVal) { toast("Datum angeben"); return; }

    await supabaseClient.from('dates').insert({

      id: dateVal, season: season, label: label, time: from + "–" + to + " Uhr", signups: [], history: [{ ts: nowIso(), text: "Termin angelegt" }]

    });

    toast("Hinzugefügt");

  }


  function openSignupModal(dateId) {

    pendingSignupDateId = dateId;

    var ident = getMyIdentity();

    document.getElementById("signupName").value = ident.name;

    document.getElementById("signupPhone").value = ident.phone;

    document.getElementById("signupModalBackdrop").classList.add("open");

  }


  async function confirmSignup() {

    var name = document.getElementById("signupName").value.trim();

    var phone = document.getElementById("signupPhone").value.trim();

    if (!name) return;

    var row = datesById[pendingSignupDateId];

    var signups = row.data.signups || [];

    signups.push({ name, phone, signedUpAt: nowIso() });

    var history = (row.data.history || []).concat({ ts: nowIso(), text: name + " hat sich eingetragen" });

    

    await supabaseClient.from('dates').update({ signups, needs_replacement: false, history }).eq('id', pendingSignupDateId);

    setMyIdentity(name, phone);

    document.getElementById("signupModalBackdrop").classList.remove("open");

    toast("Erfolgreich eingetragen!");

  }


  function openWithdrawModal(dateId, name) {

    pendingWithdraw = { dateId, name };

    document.getElementById("withdrawModalBackdrop").classList.add("open");

  }


  async function confirmWithdrawPlain() {

    var row = datesById[pendingWithdraw.dateId];

    var signups = (row.data.signups || []).filter(s => norm(s.name) !== norm(pendingWithdraw.name));

    var history = (row.data.history || []).concat({ ts: nowIso(), text: pendingWithdraw.name + " hat sich ausgetragen" });

    

    await supabaseClient.from('dates').update({ signups, needs_replacement: true, replacement_note: "zuletzt von " + pendingWithdraw.name, history }).eq('id', pendingWithdraw.dateId);

    document.getElementById("withdrawModalBackdrop").classList.remove("open");

    toast("Ausgetragen.");

  }
